Python
import hashlib, hmac, time, uuid, requests

BASE, APP_KEY, SECRET = "https://your-server", "ak_xxx", "sk_xxx"
method, path, body = "GET", "/openapi/v1/docs", ""
ts, nonce = str(int(time.time())), uuid.uuid4().hex
sts = "\n".join([APP_KEY, method, path, ts, nonce,
                 hashlib.sha256(body.encode()).hexdigest()])
sign = hmac.new(SECRET.encode(), sts.encode(), hashlib.sha256).hexdigest()
print(requests.get(BASE + path, params={"page": 1, "size": 15}, headers={
    "X-App-Key": APP_KEY, "X-Timestamp": ts, "X-Nonce": nonce, "X-Signature": sign,
}).json())
Node.js
const crypto = require("crypto");
const BASE = "https://your-server", APP_KEY = "ak_xxx", SECRET = "sk_xxx";
const method = "GET", path = "/openapi/v1/docs", body = "";
const ts = String(Math.floor(Date.now() / 1000));
const nonce = crypto.randomBytes(16).toString("hex");
const sts = [APP_KEY, method, path, ts, nonce,
  crypto.createHash("sha256").update(body).digest("hex")].join("\n");
const sign = crypto.createHmac("sha256", SECRET).update(sts).digest("hex");
fetch(`${BASE}${path}?page=1&size=15`, { headers: {
  "X-App-Key": APP_KEY, "X-Timestamp": ts, "X-Nonce": nonce, "X-Signature": sign,
}}).then((r) => r.json()).then(console.log);
